from dataclasses import dataclass

@dataclass(frozen=True)
class Instrument:
    symbol: str
    provider_symbol: str
    exchange: str
    segment: str
    name: str

# Development instrument master. Provider symbols use Yahoo Finance notation.
INDIAN_INSTRUMENTS = {
    "RELIANCE": Instrument("RELIANCE", "RELIANCE.NS", "NSE", "EQUITY", "Reliance Industries"),
    "TCS": Instrument("TCS", "TCS.NS", "NSE", "EQUITY", "Tata Consultancy Services"),
    "INFY": Instrument("INFY", "INFY.NS", "NSE", "EQUITY", "Infosys"),
    "HDFCBANK": Instrument("HDFCBANK", "HDFCBANK.NS", "NSE", "EQUITY", "HDFC Bank"),
    "ICICIBANK": Instrument("ICICIBANK", "ICICIBANK.NS", "NSE", "EQUITY", "ICICI Bank"),
    "SBIN": Instrument("SBIN", "SBIN.NS", "NSE", "EQUITY", "State Bank of India"),
    "NIFTY50": Instrument("NIFTY50", "^NSEI", "NSE", "INDEX", "NIFTY 50"),
    "BANKNIFTY": Instrument("BANKNIFTY", "^NSEBANK", "NSE", "INDEX", "NIFTY Bank"),
}

def get_instrument(symbol: str) -> Instrument:
    key = symbol.strip().upper()
    if key not in INDIAN_INSTRUMENTS:
        raise KeyError(f"Unsupported development instrument: {key}")
    return INDIAN_INSTRUMENTS[key]
